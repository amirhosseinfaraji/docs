To communicate with the native application, you can use the following methods in your web application:

### Native Data

<ParamField path="Param" type="(data: WebviewInputModel) => {}"></ParamField>

```typescript Example.ts lines
  getFlutterData(callback: (data: WebviewInputModel) => void) {
    (window as SafeAny).receiveFromFlutter = (flutterData: string) => {
      const data: WebviewInputModel = JSON.parse(flutterData);
      callback(data);
      localStorage.setItem('isWebview', 'true');
      // and set other data to your stores
    };
  }
```

### Theme Switch

<ParamField path="Param" type="theme: dark | light" required></ParamField>

```typescript Example.ts lines
switchTheme(theme: 'dark' | 'light') {
    if (!this.isWebView()) return;
    (window as SafeAny)?.darkModeHandler?.postMessage(theme);
    (window as SafeAny)?.flutter_inappwebview?.callHandler(
      'darkModeHandler',
      theme,
    );
}
```

### Get File

<ParamField path="Param" type="config: FilePickerHandlerPayload"></ParamField>

<ParamField path="Param" type="(file: File) => {}"></ParamField>

```typescript Example.ts lines
  getFile(config: FilePickerHandlerPayload, callback: (file: File) => void) {
    if (!this.isWebView()) return;
    const json = JSON.stringify(config);
    (window as SafeAny)?.uploadPhotoHandler?.postMessage(json);
    (window as SafeAny)?.flutter_inappwebview?.callHandler(
      'uploadPhotoHandler',
      json,
    );
    (window as SafeAny).onFlutterPhotoUpload = async (base64Image: string) => {
      const file = await base64ToFile(
        base64Image,
        `${new Date().getTime()}.png`,
      );
      callback(file);
    };
  }
```
